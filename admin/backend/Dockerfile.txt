FROM node:18-bullseye

# 1. Skip Puppeteer's automatic Chrome download (Saves time & storage)
ENV PUPPETEER_SKIP_CHROMIUM_DOWNLOAD=true

# 2. Install Chromium and required Linux fonts for check.js image generation
RUN apt-get update && apt-get install -y \
    chromium \
    fonts-liberation \
    fontconfig \
    --no-install-recommends && \
    rm -rf /var/lib/apt/lists/*

# Set up working directory
WORKDIR /app

# Copy package files and install dependencies
COPY package*.json ./
RUN npm install

# 3. Fixed syntax typo: Copy the rest of your application code
COPY . .

# Give the 'node' user permission to write the WhatsApp session folder
RUN chown -R node:node /app
USER node

# Expose the Hugging Face port
EXPOSE 7860

# Start the server
CMD ["node", "server.js"]
