service cloud.firestore {
  match /databases/{database}/documents {

    // --- HELPER FUNCTIONS ---
    
    function isStaff() {
      return request.auth != null;
    }

    function isAdmin() {
      return request.auth != null && request.auth.uid == 'UnNJpdzPXpNRSrfSFESRV845od83';
    }

    // --- MAIN RULES ---

    // 1. MAIN COLLECTION (The Vault)
    match /customers/{customerId} {
      // 🔓 OPEN: Anyone (logged in or out) can now read the customers collection
      allow read: if true || isStaff(); 
      
      // ONLY the Admin can add, edit, or delete main records
      allow write: if isAdmin();
    }

    // 2. TEMP ENTRIES (Agent Drafts & Admin Review)
    match /temp_entries/{entryId} {
      // Only logged-in staff can see the temporary drafts
      allow read: if isStaff();
      
      // 🔧 UPGRADED: Forces the agent to attach their own UID when creating a draft (Now strictly checks for 'staffUid' to match your frontend)
      allow create: if isStaff() && request.resource.data.staffUid == request.auth.uid;
      
      // An agent can only edit/delete if they created it, OR if the Admin is doing it (Now strictly checks 'staffUid')
      allow update, delete: if isAdmin() || request.auth.uid == resource.data.staffUid; 
    }
    
  }
}
